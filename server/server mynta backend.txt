
const express = require("express");
const cors = require("cors");
const dotenv = require("dotenv");
const OpenAI = require("openai");

dotenv.config();

const app = express();
const PORT = 8787;

app.use(cors({
  origin: [
    "http://localhost:5173",
    "http://127.0.0.1:5173",
    "http://localhost:5174",
    "http://127.0.0.1:5174"
  ]
}));

app.use(express.json());

app.get("/api/health", (req, res) => {
  res.json({
    ok: true,
    configured: Boolean(process.env.OPENAI_API_KEY)
  });
});

app.post("/api/recommendations", async (req, res) => {
  if (!process.env.OPENAI_API_KEY) {
    return res.status(503).json({
      error: "OpenAI API key is missing. Configure it in server/.env."
    });
  }

  const products = req.body.products;

  if (!Array.isArray(products) || products.length === 0) {
    return res.status(400).json({
      error: "Please add products to your Myntra catalogue first."
    });
  }

  try {
    const openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY
    });

    const response = await openai.chat.completions.create({
      model: "gpt-4.1-mini",
      messages: [
        {
          role: "system",
          content: "You are a fashion stylist for a Myntra shopping catalogue. Recommend suitable outfits from the supplied products. Use only products in the catalogue. Return concise, friendly recommendations in JSON format with a recommendations array."
        },
        {
          role: "user",
          content: JSON.stringify({
            request: req.body.prompt || "Recommend a stylish outfit.",
            products: products.slice(0, 50)
          })
        }
      ],
      response_format: { type: "json_object" }
    });

    res.json(JSON.parse(response.choices[0].message.content));
  } catch (error) {
    console.error("AI recommendation error:", error.message);
    res.status(500).json({
      error: "AI recommendations failed. Check the server terminal and API key."
    });
  }
});

app.listen(PORT, () => {
  console.log(`Myntra AI backend running at http://localhost:${PORT}`);
});
